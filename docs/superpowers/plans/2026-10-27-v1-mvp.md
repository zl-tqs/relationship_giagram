# V1 MVP 功能开发 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 实现可上线的小说人物关系图工具 V1 MVP：画布编辑（角色/关系/样式）+ 文件管理（云端自动保存 + PC 本地保存）+ 撤销重做 + 基础搜索 + 导出 + 快捷键。

**Architecture:** 前端 React Flow 画布通过 REST API 与 NestJS 后端交互；后端 TypeORM + PostgreSQL 持久化 `users / projects / canvases`（画布节点与边以 JSONB 存储）。V1 角色与关系内嵌于画布 JSONB，不建结构化表（结构化表随 V2 能力/属性/标签引入，见 spec 第 8 章版本映射）。

**Tech Stack:** React 18 + TS 6 + Vite 8 + React Flow 12 + Zustand 5 + Ant Design 5 + react-hotkeys-hook 4 + react-router-dom 7 + axios；NestJS 12 + TypeORM + PostgreSQL 16+ + @nestjs/jwt + argon2；Tauri 2（fs/dialog 插件实现 PC 本地保存）。

**Spec:** `小说人物关系图工具-技术设计文档.md`（项目根目录；本计划实现其中第 8 章"第一版 MVP"清单 + 第 6.1 章 V1 端点 + 第 4.2 章 users/projects/canvases 表）

## Global Constraints

- 包管理器 npm；项目根目录 `D:\relationship_giagram`；所有命令以该目录为基准
- 后端 API 全局前缀 `api/v1`；统一错误体 `{ code, message, details }`

- [ ] **Step 3: auth.service**

`backend/src/auth/auth.service.ts`：

```typescript
import { ConflictException, Injectable, UnauthorizedException } from '@nestjs/common';
import { JwtService } from '@nestjs/jwt';
import * as argon2 from 'argon2';
import { UsersService } from '../users/users.service.js';
import { LoginDto } from './dto/login.dto.js';
import { RegisterDto } from './dto/register.dto.js';

@Injectable()
export class AuthService {
  constructor(
    private readonly users: UsersService,
    private readonly jwt: JwtService,
  ) {}

  async register(dto: RegisterDto) {
    if (await this.users.findByEmail(dto.email)) {
      throw new ConflictException({ code: 'EMAIL_TAKEN', message: '邮箱已被注册' });
    }
    const hash = await argon2.hash(dto.password);
    const user = await this.users.create(dto.email, hash, dto.nickname);
    return { accessToken: this.sign(user.id), user: this.public(user) };
  }

  async login(dto: LoginDto) {
    const user = await this.users.findByEmail(dto.email);
    if (!user?.passwordHash || !(await argon2.verify(user.passwordHash, dto.password))) {
      throw new UnauthorizedException({ code: 'BAD_CREDENTIALS', message: '邮箱或密码错误' });
    }
    return { accessToken: this.sign(user.id), user: this.public(user) };
  }

  async anonymous(nickname?: string) {
    const user = await this.users.createAnonymous(nickname);
    return { accessToken: this.sign(user.id), user: this.public(user) };
  }

  private sign(userId: string) {
    return this.jwt.sign({ sub: userId }, { expiresIn: '15m' });
  }

  private public(user: { id: string; email: string | null; nickname: string; isAnonymous: boolean }) {
    const { id, email, nickname, isAnonymous } = user;
    return { id, email, nickname, isAnonymous };
  }
}
```

```typescript
import { CanActivate, ExecutionContext, Injectable, UnauthorizedException } from '@nestjs/common';
import { JwtService } from '@nestjs/jwt';

@Injectable()
export class JwtAuthGuard implements CanActivate {
  constructor(private readonly jwt: JwtService) {}

  canActivate(context: ExecutionContext): boolean {
    const req = context.switchToHttp().getRequest();
    const auth: string | undefined = req.headers.authorization;
    if (!auth?.startsWith('Bearer ')) {
      throw new UnauthorizedException({ code: 'NO_TOKEN', message: '未登录' });
    }
    try {
      const payload = this.jwt.verify(auth.slice(7));
      req.user = { userId: payload.sub };
      return true;
    } catch {
      throw new UnauthorizedException({ code: 'BAD_TOKEN', message: '令牌无效或已过期' });
    }
  }
}
```

`backend/src/projects/projects.service.ts`：

```typescript
import { ForbiddenException, Injectable, NotFoundException } from '@nestjs/common';
import { InjectRepository } from '@nestjs/typeorm';
import { Repository } from 'typeorm';
import { Canvas } from '../canvases/canvas.entity.js';
import { Project } from './project.entity.js';
import { CreateProjectDto } from './dto/create-project.dto.js';
import { UpdateProjectDto } from './dto/update-project.dto.js';

@Injectable()
export class ProjectsService {
  constructor(
    @InjectRepository(Project) private readonly projects: Repository<Project>,
    @InjectRepository(Canvas) private readonly canvases: Repository<Canvas>,
  ) {}

  async recent(userId: string) {
    return this.projects.find({
      where: { ownerId: userId },
      order: { lastModifiedAt: 'DESC' },
      take: 20,
      select: ['id', 'name', 'description', 'lastModifiedAt'],
    });
  }

  async create(userId: string, dto: CreateProjectDto) {
    const project = await this.projects.save(this.projects.create({ ownerId: userId, name: dto.name }));
    await this.canvases.save(this.canvases.create({ projectId: project.id }));
    return project;
  }

  async findOne(userId: string, id: string) {
    const project = await this.projects.findOne({ where: { id } });
    if (!project) throw new NotFoundException({ code: 'NOT_FOUND', message: '项目不存在' });
    if (project.ownerId !== userId) throw new ForbiddenException({ code: 'FORBIDDEN', message: '无权访问该项目' });
    return project;
  }

  async update(userId: string, id: string, dto: UpdateProjectDto) {
    const project = await this.findOne(userId, id);
    Object.assign(project, dto);
    project.lastModifiedAt = new Date();
    return this.projects.save(project);
  }

  async remove(userId: string, id: string) {
    const project = await this.findOne(userId, id);
    return this.projects.softDelete(project.id);
  }
}
```

```typescript
import { Body, Controller, Delete, Get, Param, Patch, Post, UseGuards } from '@nestjs/common';
import { JwtAuthGuard } from '../auth/jwt-auth.guard.js';
```

```typescript
import { ConflictException, Injectable, NotFoundException } from '@nestjs/common';
import { InjectRepository } from '@nestjs/typeorm';
import { Repository } from 'typeorm';
import { ProjectsService } from '../projects/projects.service.js';
import { Canvas } from './canvas.entity.js';
import { SaveCanvasDto } from './dto/save-canvas.dto.js';

@Injectable()
export class CanvasesService {
  constructor(
    @InjectRepository(Canvas) private readonly canvases: Repository<Canvas>,
    private readonly projects: ProjectsService,
  ) {}

  async findByProject(userId: string, projectId: string) {
    await this.projects.findOne(userId, projectId);
    const canvas = await this.canvases.findOne({ where: { projectId } });
    if (!canvas) throw new NotFoundException({ code: 'NOT_FOUND', message: '画布不存在' });
    return canvas;
  }

  async save(userId: string, projectId: string, dto: SaveCanvasDto) {
    const canvas = await this.findByProject(userId, projectId);
    if (canvas.version !== dto.baseVersion) {
      throw new ConflictException({
        code: 'VERSION_CONFLICT',
        message: '画布已被其他端修改',
        details: { serverVersion: canvas.version, clientVersion: dto.baseVersion },
      });
    }
    canvas.nodes = dto.nodes;
    canvas.edges = dto.edges;
    canvas.viewport = dto.viewport;
    canvas.version += 1;
    const saved = await this.canvases.save(canvas);
    await this.projects.update(userId, projectId, {});
    return { id: saved.id, version: saved.version };
  }
}
```

```typescript
export const api = axios.create({ baseURL: '/api/v1' });

api.interceptors.request.use((config) => {
  const token = localStorage.getItem('accessToken');
  if (token) config.headers.Authorization = `Bearer ${token}`;
  return config;
});

api.interceptors.response.use(
  (res) => res,
  (err) => {
    if (err.response?.status === 401 && localStorage.getItem('accessToken')) {
      localStorage.removeItem('accessToken');
      window.location.href = '/login';
    }
    return Promise.reject(err);
  },
);
```

```typescript
import { create } from 'zustand';
import * as authApi from '../api/auth.js';

interface AuthState {
  token: string | null;
  user: { id: string; nickname: string; isAnonymous: boolean } | null;
  login: (email: string, password: string) => Promise<void>;
  register: (email: string, password: string, nickname: string) => Promise<void>;
  anonymous: () => Promise<void>;
  logout: () => void;
}

export const useAuthStore = create<AuthState>((set) => ({
  token: localStorage.getItem('accessToken'),
  user: null,
  async login(email, password) {
    const res = await authApi.login({ email, password });
    localStorage.setItem('accessToken', res.accessToken);
    set({ token: res.accessToken, user: res.user });
  },
  async register(email, password, nickname) {
    const res = await authApi.register({ email, password, nickname });
    localStorage.setItem('accessToken', res.accessToken);
    set({ token: res.accessToken, user: res.user });
  },
  async anonymous() {
    const res = await authApi.anonymous();
    localStorage.setItem('accessToken', res.accessToken);
    set({ token: res.accessToken, user: res.user });
  },
  logout() {
    localStorage.removeItem('accessToken');
    set({ token: null, user: null });
  },
}));
```

```typescript
import { create } from 'zustand';
import { applyNodeChanges, applyEdgeChanges, type Node, type Edge, type NodeChange, type EdgeChange } from '@xyflow/react';
import type { CharacterData, RelationData } from '../types';

const HISTORY_LIMIT = 100;

type HistoryEntry = { nodes: Node[]; edges: Edge[] };

interface CanvasStore {
  nodes: Node<CharacterData>[];
  edges: Edge<RelationData>[];
  viewport: { x: number; y: number; zoom: number };
  version: number;
  past: HistoryEntry[];
  future: HistoryEntry[];
  addCharacter: (shape: 'circle' | 'square') => void;
  removeNode: (id: string) => void;
  onNodesChange: (changes: NodeChange[]) => void;
  onEdgesChange: (changes: EdgeChange[]) => void;
  undo: () => void;
  redo: () => void;
  canUndo: () => boolean;
  canRedo: () => boolean;
}

let seq = 0;
const nextId = () => `n${++seq}`;

function commit(set: (partial: Partial<CanvasStore>) => void, get: () => CanvasStore) {
  const { nodes, edges, past } = get();
  if (past.length >= HISTORY_LIMIT) past.shift();
  set({ past: [...past, { nodes, edges }], future: [] });
}

export const useCanvasStore = create<CanvasStore>((set, get) => ({
  nodes: [],
  edges: [],
  viewport: { x: 0, y: 0, zoom: 1 },
  version: 1,
  past: [],
  future: [],

  addCharacter: (shape) => {
    commit(set, get);
    const id = nextId();
    const node: Node<CharacterData> = {
      id,
      type: 'character',
      position: { x: 100 + Math.random() * 300, y: 100 + Math.random() * 200 },
      data: { shape, name: '新角色', gender: '未知' },
    };
    set({ nodes: [...get().nodes, node] });
  },

  removeNode: (id) => {
    commit(set, get);
    set({
      nodes: get().nodes.filter((n) => n.id !== id),
      edges: get().edges.filter((e) => e.source !== id && e.target !== id),
    });
  },

  onNodesChange: (changes) => {
    set({ nodes: applyNodeChanges(changes, get().nodes) });
  },

  onEdgesChange: (changes) => {
    set({ edges: applyEdgeChanges(changes, get().edges) });
  },

  undo: () => {
    const { past, future, nodes, edges } = get();
    const prev = past[past.length - 1];
    if (!prev) return;
    set({
      past: past.slice(0, -1),
      future: [{ nodes, edges }, ...future],
      nodes: prev.nodes,
      edges: prev.edges,
    });
  },

  redo: () => {
    const { past, future } = get();
    const next = future[0];
    if (!next) return;
    set({
      future: future.slice(1),
      past: [...past, { nodes: get().nodes, edges: get().edges }],
      nodes: next.nodes,
      edges: next.edges,
    });
  },

  canUndo: () => get().past.length > 0,
  canRedo: () => get().future.length > 0,
}));
```

```tsx
import { Button, Space, message } from 'antd';
import { useEffect, useRef } from 'react';
import { useParams } from 'react-router-dom';
import { AppLayout } from '../components/layout/AppLayout';
import { CanvasEditor } from '../components/canvas/CanvasEditor';
import { useCanvasStore } from '../stores/canvasStore';
import { loadCanvas, saveCanvas } from '../api/canvas';
import { toApiError } from '../api/client';

export function WorkspacePage() {
  const { projectId } = useParams<{ projectId: string }>();
  const timer = useRef<number>();

  useEffect(() => {
    if (!projectId) return;
    loadCanvas(projectId)
      .then((c) => useCanvasStore.setState({ nodes: c.nodes, edges: c.edges, viewport: c.viewport, version: c.version }))
      .catch(() => message.error('加载画布失败'));

    timer.current = window.setInterval(() => {
      const s = useCanvasStore.getState();
      saveCanvas(projectId, {
        nodes: s.nodes, edges: s.edges, viewport: s.viewport, version: s.version,
      })
        .then((r) => useCanvasStore.setState({ version: r.version }))
        .catch((e) => {
          if (toApiError(e).code === 'VERSION_CONFLICT') message.warning('画布已在其他端修改，请刷新');
          else message.error('自动保存失败');
        });
    }, 3 * 60 * 1000);

    return () => clearInterval(timer.current);
  }, [projectId]);

  const onSave = async () => {
    if (!projectId) return;
    const s = useCanvasStore.getState();
    try {
      const r = await saveCanvas(projectId, {
        nodes: s.nodes, edges: s.edges, viewport: s.viewport, version: s.version,
      });
      useCanvasStore.setState({ version: r.version });
      message.success('已保存');
    } catch (e) {
      message.error(toApiError(e).message);
    }
  };

  return (
    <AppLayout>
      <CanvasEditor />
      <Space style={{ position: 'absolute', top: 8, right: 8, zIndex: 10 }}>
        <Button onClick={onSave}>保存</Button>
      </Space>
    </AppLayout>
  );
}
```

```typescript
import { save, open } from '@tauri-apps/plugin-dialog';
import { writeTextFile, readTextFile } from '@tauri-apps/plugin-fs';
import type { CanvasState } from '../../types';

export const isDesktop = () => typeof window !== 'undefined' && '__TAURI_INTERNALS__' in window;

export async function saveLocal(state: CanvasState) {
  if (isDesktop()) {
    const path = await save({ filters: [{ name: '关系图', extensions: ['rmap'] }] });
    if (!path) return;
    await writeTextFile(path, JSON.stringify(state, null, 2));
  } else {
    const blob = new Blob([JSON.stringify(state, null, 2)], { type: 'application/json' });
    const a = document.createElement('a');
    a.href = URL.createObjectURL(blob);
    a.download = '关系图.rmap';
    a.click();
    URL.revokeObjectURL(a.href);
  }
}

export async function loadLocal(): Promise<CanvasState | null> {
  if (isDesktop()) {
    const path = await open({ filters: [{ name: '关系图', extensions: ['rmap'] }] });
    if (!path) return null;
    return JSON.parse(await readTextFile(path));
  }
  return null;
}
```
